alter proc [dbo].[sp_GetExpences]  
(  
@SBranchID int,  
@Month int,  
@Year int,  
@ExpenceType int  
)  
AS  
BEGIN  
 select ExpenceTypeID as ID, ExpenceTypeName as Name from [dbo].[ExpenceTypeMaster] where userid=0 or userid=@SBranchID 

 if(@ExpenceType<>0)  
 begin  
 Select ExpenceID ,[ReferanceNumber] ,[ExpenceTypeID] ,[PaymentMode] ,[Remark]  
      ,(select sum(Amount) from [dbo].[ExpenceDetails] PD where PD.[ExpenceID] = PM.[ExpenceID]) as[PaymentAmount] ,[PaymentStatus] ,[Year] ,[Month] ,[PaymentDate] ,[SBranchID]  
      ,[ExpenceTitle] ,[EnteredDate]  
  FROM [dbo].[ExpenceMaster] PM  
  where SBranchID=@SBranchID and Month=@Month and Year=@Year  and ExpenceTypeID=@ExpenceType  and    SBranchID=@SBranchID 
  order by [PaymentDate] desc  
  end  
  else  
  begin  
 Select ExpenceID ,[ReferanceNumber] ,PM.[ExpenceTypeID] ,[PaymentMode] ,[Remark]  
      ,(select sum(Amount) from [dbo].[ExpenceDetails] PD where PD.[ExpenceID] = PM.[ExpenceID]) as[PaymentAmount] ,[PaymentStatus] ,[Year] ,[Month] ,[PaymentDate] ,[SBranchID]  
      ,[ExpenceTitle] ,[EnteredDate],ETM.ExpenceTypeName as ExpenceType,ETM.ChartColor  
         
  FROM [dbo].[ExpenceMaster] PM left outer join [ExpenceTypeMaster] ETM on ETM.ExpenceTypeID= PM.ExpenceTypeID   
  where SBranchID=@SBranchID and Month=@Month and Year=@Year  and  (SBranchID=@SBranchID or ETM.Userid=0) 
  order by [PaymentDate] desc  
  end  
END  
  go
