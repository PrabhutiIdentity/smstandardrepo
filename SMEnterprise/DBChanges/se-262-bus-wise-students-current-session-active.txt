
  CREATE OR ALTER PROCEDURE dbo.sp_GetBusWiseStudents
    @SBranchID INT,
    @BusID INT = 0,
    @ShiftID INT = 0,
    @Type INT = 0
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @CurrentDate DATE = CAST(GETDATE() AS DATE);
    DECLARE @SelectedBusID INT = ISNULL(@BusID, 0);

    IF ISNULL(@SelectedBusID, 0) = 0
    BEGIN
        SELECT TOP 1 @SelectedBusID = ISNULL(VD.VehicleID, 0)
        FROM dbo.VehicleDetails VD
        WHERE ISNULL(VD.SBranchID, 0) = @SBranchID
        ORDER BY VD.VehicleNumber, VD.VehicleID;
    END

    ;WITH CurrentSession AS
    (
        SELECT TOP 1 SM.SessionID
        FROM dbo.SessionMaster SM
        WHERE ISNULL(SM.SBranchID, 0) = @SBranchID
          AND ISNULL(SM.SessionStatus, 0) = 1
        ORDER BY SM.SessionStartDate DESC, SM.SessionID DESC
    ),
    ActiveTransport AS
    (
        SELECT
            THAD.UserID AS StudentID,
            THAD.KeyID AS StopID,
            ROW_NUMBER() OVER
            (
                PARTITION BY THAD.UserID
                ORDER BY ISNULL(THAD.StartDate, '19000101') DESC, THAD.THChangeID DESC
            ) AS RowNo
        FROM dbo.TransportHostalAllocationDelocation THAD
        WHERE ISNULL(THAD.UserType, 0) = 0
          AND ISNULL(THAD.ChangeType, 0) = 0
          AND ISNULL(THAD.KeyID, 0) <> 0
          AND CAST(ISNULL(THAD.StartDate, @CurrentDate) AS DATE) <= @CurrentDate
          AND (
              THAD.EndDate IS NULL
              OR YEAR(THAD.EndDate) = 1
              OR CAST(THAD.EndDate AS DATE) >= @CurrentDate
          )
    ),
    StudentTransport AS
    (
        SELECT
            SM.StudentID,
            ISNULL(NULLIF(SM.VehicleRouteID, 0), 0) AS VehicleRouteID,
            ISNULL(NULLIF(SM.StopID, 0), AT.StopID) AS StopID
        FROM dbo.StudentMaster SM
        LEFT JOIN ActiveTransport AT
            ON AT.StudentID = SM.StudentID
           AND AT.RowNo = 1
        WHERE ISNULL(SM.SBranchID, 0) = @SBranchID
    )
    SELECT DISTINCT
        SM.StudentID,
        SS.StudentSessionUID,
        ISNULL(SM.SchoolUID, '') AS SchoolUID,
         'STUD'+RIGHT(REPLICATE('0',6)+CAST(SM.StudentID AS VARCHAR(6)),6)  AS StudentSID,
        ISNULL(SM.Name, '') AS Name,
        ISNULL(SM.Gender, 0) AS Gender,
        ISNULL(PM.FatherName, '') AS FatherName,
        ISNULL(PM.MotherName, '') AS MotherName,
        ISNULL(SM.GuardianMobileNo, '') AS GuardianMobileNo,
        ISNULL(PM.FatherMobileNo, '') AS FatherMobileNo,
        ISNULL(PM.MotherMobileNo, '') AS MotherMobileNo,
        ISNULL(CM.SequenceNo, 0) AS SequenceNo,
        ISNULL(CM.ClassName, '') AS ClassName,
        ISNULL(CLS.Name, '') AS SectionName
    FROM StudentTransport ST
    INNER JOIN dbo.StudentMaster SM
        ON SM.StudentID = ST.StudentID
    LEFT JOIN dbo.Transport_Vehicle_Route_Details TVRD
        ON TVRD.StopID = ST.StopID
       AND ISNULL(ST.VehicleRouteID, 0) = 0
    INNER JOIN dbo.Transport_Vehicle_Route TVR
        ON TVR.VehicleRouteID = ISNULL(NULLIF(ST.VehicleRouteID, 0), TVRD.VehicleRouteID)
    INNER JOIN dbo.Student_Session SS
        ON SS.StudentID = SM.StudentID
       AND ISNULL(SS.SBranchID, 0) = @SBranchID
       AND ISNULL(SS.Status, 0) = 1
       AND SS.SessionID = (SELECT TOP 1 SessionID FROM CurrentSession)
    LEFT JOIN dbo.ParentMaster PM
        ON PM.ParentID = SM.ParentID
    LEFT JOIN dbo.ClassMaster CM
        ON CM.ClassID = SS.ClassID
    LEFT JOIN dbo.Class_Sections CLS
        ON CLS.ID = SS.SectionID
    WHERE ISNULL(SM.SBranchID, 0) = @SBranchID
      AND ISNULL(TVR.SBranchID, 0) = @SBranchID
      AND ISNULL(TVR.VehicleID, 0) = @SelectedBusID
      AND (@ShiftID = 0 OR ISNULL(TVR.ShiftType, 0) = @ShiftID)
    ORDER BY SequenceNo, ClassName, SectionName, Name;

    SELECT
        VD.VehicleID AS ID,
        ISNULL(VD.VehicleNumber, '') AS Name
    FROM dbo.VehicleDetails VD
    WHERE ISNULL(VD.SBranchID, 0) = @SBranchID
    ORDER BY VD.VehicleNumber, VD.VehicleID;

    SELECT ISNULL(@SelectedBusID, 0) AS ID;

    SELECT
        SBM.SBranchID,
        SBM.BranchName,
        SBM.Logo,
        SBM.ContactNo,
        SBM.EmailID,
        SBM.Address,
        SBM.PrincipalName,
        SBM.PrincipalMobile,
        SBM.PrincipalEmail,
        SBM.BranchSchoolName,
        SBM.TagLine
    FROM dbo.SBranchMaster SBM
    WHERE SBM.SBranchID = @SBranchID;
END
GO
