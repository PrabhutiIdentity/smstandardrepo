ALTER PROCEDURE [dbo].[spn_InsertUpdateStudentSessionDetails]
(
    @StudentSessionUID int,
    @StudentID int,
    @ClassID int,
    @SectionID int,
    @FromDate date,
    @ToDate date,
    @Status int,
    @RollNo nvarchar(50),
    @SBranchID int,
    @QuotaID int,
    @FeePaymentMode int,
    @SessionID int = 1,
    @HouseID int = 1,
    @OptedSubjects ut_Name_ID_Utility READONLY,
    @IsAdmissionFeeApplicable int,
    @ReasonforInactive nvarchar(max),
    @IsNewSessionRequest bit = 0
)
AS
BEGIN
    SET NOCOUNT ON;

    IF (@IsNewSessionRequest = 1)
    BEGIN
        IF EXISTS
        (
            SELECT 1
            FROM dbo.Student_Session SS
            WHERE SS.StudentID = @StudentID
              AND SS.SBranchID = @SBranchID
              AND SS.SessionID = @SessionID
              AND @FromDate <= SS.ToDate
              AND @ToDate >= SS.FromDate
        )
        BEGIN
            SELECT -1;
            RETURN;
        END;

        IF (@Status = 1)
        BEGIN
            UPDATE dbo.Student_Session
            SET [Status] = 0
            WHERE StudentID = @StudentID
              AND SBranchID = @SBranchID;
        END;

        INSERT INTO dbo.Student_Session
        (
            StudentID, ClassID, SectionID, FromDate, ToDate, [Status],
            RollNo, SBranchID, QuotaID, FeePaymentMode, SessionID,
            HouseID, IsAdmissionFeeApplicable
        )
        VALUES
        (
            @StudentID, @ClassID, @SectionID, @FromDate, @ToDate, @Status,
            @RollNo, @SBranchID, @QuotaID, @FeePaymentMode, @SessionID,
            @HouseID, @IsAdmissionFeeApplicable
        );

        SELECT @StudentSessionUID = CAST(SCOPE_IDENTITY() AS int);
        SELECT @StudentID;
    END
    ELSE
    BEGIN
        IF (@Status = 1)
        BEGIN
            UPDATE dbo.Student_Session
            SET [Status] = 0
            WHERE StudentID = @StudentID
              AND SBranchID = @SBranchID
              AND StudentSessionUID <> @StudentSessionUID;
        END;

        UPDATE dbo.Student_Session
        SET
            ClassID = @ClassID,
            SectionID = @SectionID,
            HouseID = @HouseID,
            FromDate = @FromDate,
            ToDate = @ToDate,
            [Status] = @Status,
            RollNo = @RollNo,
            SBranchID = @SBranchID,
            QuotaID = @QuotaID,
            FeePaymentMode = @FeePaymentMode,
            SessionID = @SessionID,
            IsAdmissionFeeApplicable = @IsAdmissionFeeApplicable,
            ReasonforInactive = @ReasonforInactive
        WHERE SBranchID = @SBranchID
          AND StudentSessionUID = @StudentSessionUID
          AND StudentID = @StudentID;

        SELECT @StudentID;
    END;

    DELETE FROM dbo.Student_Session_OptionalSubjects
    WHERE StudentSessionID = @StudentSessionUID;

    INSERT INTO dbo.Student_Session_OptionalSubjects (StudentSessionID, OpSubjectID)
    SELECT @StudentSessionUID, ID
    FROM @OptedSubjects
    WHERE Extra2 = 1;
END

CREATE OR ALTER PROCEDURE usp_GetLatestStudentSessionUID
    @StudentID INT,
    @SBranchID INT,
    @SessionID INT,
    @ClassID INT,
    @SectionID INT
AS
BEGIN
    SET NOCOUNT ON;

    SELECT TOP 1 StudentSessionUID
    FROM dbo.Student_Session
    WHERE StudentID = @StudentID
      AND SBranchID = @SBranchID
      AND SessionID = @SessionID
      AND ClassID = @ClassID
      AND SectionID = @SectionID
    ORDER BY StudentSessionUID DESC;
END

CREATE OR ALTER PROCEDURE usp_CopyOptionalSubjectsFromPreviousMatchingSession
    @TargetStudentSessionUID INT
AS
BEGIN
    SET NOCOUNT ON;

    -- 1. Check if targets already exist to prevent duplicate copies
    IF EXISTS
    (
        SELECT 1
        FROM dbo.Student_Session_OptionalSubjects
        WHERE StudentSessionID = @TargetStudentSessionUID
    )
    BEGIN
        RETURN;
    END;

    DECLARE @StudentID INT,
            @SBranchID INT,
            @ClassID INT,
            @SectionID INT,
            @FromDate DATE,
            @EducationLevelID INT,
            @GroupID INT,
            @SourceStudentSessionUID INT;

    -- 2. Fetch basic session tracking details
    SELECT
        @StudentID = SS.StudentID,
        @SBranchID = SS.SBranchID,
        @ClassID = SS.ClassID,
        @SectionID = SS.SectionID,
        @FromDate = SS.FromDate
    FROM dbo.Student_Session SS
    WHERE SS.StudentSessionUID = @TargetStudentSessionUID;

    -- 3. Extract Meta filters for mapping comparison
    SELECT @EducationLevelID = CM.EducationLevelID
    FROM dbo.ClassMaster CM
    WHERE CM.ClassID = @ClassID;

    SELECT @GroupID = CS.GroupID
    FROM dbo.Class_Sections CS
    WHERE CS.ID = @SectionID;

    -- 4. Find the most appropriate source timeline session
    SELECT TOP 1 @SourceStudentSessionUID = PSS.StudentSessionUID
    FROM dbo.Student_Session PSS
    INNER JOIN dbo.ClassMaster PCM ON PCM.ClassID = PSS.ClassID
    INNER JOIN dbo.Class_Sections PCS ON PCS.ID = PSS.SectionID
    INNER JOIN dbo.Student_Session_OptionalSubjects PSOS ON PSOS.StudentSessionID = PSS.StudentSessionUID
    WHERE PSS.StudentID = @StudentID
      AND PSS.SBranchID = @SBranchID
      AND PSS.StudentSessionUID <> @TargetStudentSessionUID
      AND ISNULL(PCM.EducationLevelID, 0) = ISNULL(@EducationLevelID, 0)
      AND ISNULL(PCS.GroupID, 0) = ISNULL(@GroupID, 0)
      AND PSS.FromDate <= @FromDate
    ORDER BY PSS.ToDate DESC, PSS.StudentSessionUID DESC;

    -- 5. Terminate process safely if no match history found
    IF ISNULL(@SourceStudentSessionUID, 0) = 0
    BEGIN
        RETURN;
    END;

    -- 6. Direct bulk migration for optional mapped structures
    INSERT INTO dbo.Student_Session_OptionalSubjects (StudentSessionID, OpSubjectID)
    SELECT DISTINCT @TargetStudentSessionUID, PSOS.OpSubjectID
    FROM dbo.Student_Session_OptionalSubjects PSOS
    INNER JOIN dbo.SubjectMasterT SMT
        ON SMT.SubjectID = PSOS.OpSubjectID
       AND ISNULL(SMT.GroupID, 0) = ISNULL(@GroupID, 0)
       AND ISNULL(SMT.IsOptionalsubject, 0) = 1
       AND ISNULL(SMT.Status, 1) = 1
    WHERE PSOS.StudentSessionID = @SourceStudentSessionUID
      AND NOT EXISTS
      (
          SELECT 1
          FROM dbo.Student_Session_OptionalSubjects TSOS
          WHERE TSOS.StudentSessionID = @TargetStudentSessionUID
            AND TSOS.OpSubjectID = PSOS.OpSubjectID
      );
END
