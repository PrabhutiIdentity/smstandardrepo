IF OBJECT_ID('dbo.usp_GetActiveTransportFlags', 'P') IS NOT NULL
    DROP PROCEDURE dbo.usp_GetActiveTransportFlags;
GO

CREATE PROCEDURE dbo.usp_GetActiveTransportFlags
    @StudentIDsString NVARCHAR(MAX), -- Comma-separated string jaise '1,2,3'
    @CurrentDate DATE
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @StudentIDs TABLE (StudentID INT NOT NULL PRIMARY KEY);
    DECLARE @List NVARCHAR(MAX) = ISNULL(@StudentIDsString, '') + ',';
    DECLARE @Pos INT;
    DECLARE @Value NVARCHAR(50);
    DECLARE @StudentID INT;

    WHILE LEN(@List) > 0
    BEGIN
        SET @Pos = CHARINDEX(',', @List);
        IF @Pos = 0 BREAK;

        SET @Value = LTRIM(RTRIM(LEFT(@List, @Pos - 1)));

        IF @Value <> '' AND ISNUMERIC(@Value) = 1
        BEGIN
            SET @StudentID = CAST(@Value AS INT);

            IF NOT EXISTS (SELECT 1 FROM @StudentIDs WHERE StudentID = @StudentID)
            BEGIN
                INSERT INTO @StudentIDs (StudentID) VALUES (@StudentID);
            END
        END

        SET @List = SUBSTRING(@List, @Pos + 1, LEN(@List));
    END

    SELECT 
        THAD.UserID AS StudentID,
        MAX(ISNULL(THAD.KeyID, 0)) AS StopID
    FROM dbo.TransportHostalAllocationDelocation THAD
    INNER JOIN @StudentIDs SI ON SI.StudentID = THAD.UserID
    WHERE THAD.UserType = 0
      AND ISNULL(THAD.ChangeType, 0) = 0
      AND ISNULL(THAD.KeyID, 0) <> 0
      AND CAST(THAD.StartDate AS DATE) <= @CurrentDate
      AND (
          THAD.EndDate IS NULL
          OR YEAR(THAD.EndDate) = 1
          OR CAST(THAD.EndDate AS DATE) >= @CurrentDate
      )
    GROUP BY THAD.UserID;
END

GO

IF OBJECT_ID('dbo.usp_HasPaidTransportFeeForAllocation', 'P') IS NOT NULL
    DROP PROCEDURE dbo.usp_HasPaidTransportFeeForAllocation;
GO

CREATE PROCEDURE dbo.usp_HasPaidTransportFeeForAllocation
    @StudentID INT,
    @THChangeID INT,
    @SBranchID INT
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @AllocationStart DATE,
            @AllocationEnd DATE,
            @SessionID INT;

    -- 1. Get Allocation/Delocation details
    SELECT
        @AllocationStart = CAST(THAD.StartDate AS DATE),
        @AllocationEnd = CASE
            WHEN THAD.EndDate IS NULL OR YEAR(THAD.EndDate) = 1 THEN CONVERT(DATE, '99991231', 112)
            ELSE CAST(THAD.EndDate AS DATE)
        END,
        @SessionID = ISNULL(THAD.SessionID, 0)
    FROM dbo.TransportHostalAllocationDelocation THAD
    WHERE THAD.THChangeID = @THChangeID
      AND ISNULL(THAD.UserType, 0) = 0
      AND ISNULL(THAD.ChangeType, 0) = 0
      AND THAD.UserID = @StudentID;

    -- 2. Fallback to active session if not assigned
    IF ISNULL(@SessionID, 0) = 0
    BEGIN
        SELECT TOP 1 @SessionID = SM.SessionID
        FROM dbo.SessionMaster SM
        WHERE SM.SBranchID = @SBranchID
          AND SM.SessionStatus = 1
        ORDER BY SM.SessionStartDate DESC;
    END

    -- 3. Check if fee payment exists in the range
    SELECT TOP 1 1
    FROM dbo.PaymentDetails PD
    INNER JOIN dbo.FeeTypeMaster FTM ON FTM.FeeTypeID = PD.FeeTypeID
    INNER JOIN dbo.PaymentMaster PM ON PM.PaymentID = PD.PaymentID
    INNER JOIN dbo.SessionMaster SM ON SM.SessionID = @SessionID
    WHERE @AllocationStart IS NOT NULL
      AND PD.PayeeID = @StudentID
      AND ISNULL(PD.SBranchID, 0) = @SBranchID
      AND ISNULL(FTM.SBranchID, @SBranchID) = @SBranchID
      AND FTM.FeeTypeApplicable = 5 -- Transport Fee Type Flag
      AND ISNULL(PD.PaymentRecieved, 0) > 0
      AND ISNULL(PD.PaymentStatus, 0) = 0
      AND ISNULL(PM.PaymentStatus, 0) = 0
      AND ISNULL(PM.SessionID, @SessionID) = @SessionID
      -- Year/Month conversion check for Session Window
      AND ((CONVERT(INT, PD.Year) * 100) + ISNULL(PD.Month, 0))
          BETWEEN ((YEAR(SM.SessionStartDate) * 100) + MONTH(SM.SessionStartDate))
              AND ((YEAR(SM.SessionEndDate) * 100) + MONTH(SM.SessionEndDate))
      -- Year/Month conversion check for Allocation Window
      AND ((CONVERT(INT, PD.Year) * 100) + ISNULL(PD.Month, 0))
          BETWEEN ((YEAR(@AllocationStart) * 100) + MONTH(@AllocationStart))
              AND ((YEAR(@AllocationEnd) * 100) + MONTH(@AllocationEnd));
END

go
