     
alter procedure [dbo].[spn_GetSPromotetudentsByClassSection]  --2,1,1,3      
(        
@ClassID int,        
@SectionID int,        
@SBranchID int,        
@SessionID int=0        
)        
as         
begin      
Declare @SelectedSessionEndDate date      
 Declare @Status int=1        
 if(@SessionID=0)        
 begin        
  Select @SessionID=SessionID from SessionMaster where SessionStatus=1 and SBranchID=@SBranchID         
 select @Status=SessionStatus from SessionMaster where SessionID=@SessionID        
 end        
  select @SelectedSessionEndDate=SessionEndDate from SessionMaster where SessionID=@SessionID       
      
 if(@SectionID<>0)        
 begin        
  Select SM.StudentID,'STUD'+RIGHT(REPLICATE('0',6)+CAST(SM.StudentID AS VARCHAR(6)),6) as StudentSID, Name,Photo,SS.QuotaID,        
  PM.FatherName,SM.SchoolUID,SS.FeePaymentMode,SM.Gender        
  from StudentMaster SM left outer join Student_Session SS on SS.StudentID=SM.StudentID        
  left outer join Parentmaster PM on PM.ParentID=SM.ParentID        
  where (SS.ClassID=@ClassID and SS.SectionID=@SectionID and SS.SessionID=@SessionID) and SM.SBranchID=@SBranchID and  SS.Status=1  
    
 order by Name        
 end        
 else        
 begin          
 set @ClassID=0        
   Select SM.StudentID,'STUD'+RIGHT(REPLICATE('0',6)+CAST(SM.StudentID AS VARCHAR(6)),6) as StudentSID, Name,Photo,        
  PM.FatherName,SM.SchoolUID,SM.Gender        
  from StudentMaster SM   left outer join Parentmaster PM on PM.ParentID=SM.ParentID        
   where  SM.SBranchID=@SBranchID   
   and StudentID not in (select StudentID from Student_Session SS where SS.Status=1)  
    order by Name        
 end        
         
 select ClassID,ClassName from ClassMaster where SBranchID=@SBranchID        
        
 select ID,Name from Class_Sections where ClassID=@ClassID        
        
 select @ClassID        
 select @SectionID        
        
 Select SessionName as Name,SessionID as ID,SessionStatus as Extra1 from SessionMaster where SBranchID=@SBranchID        
         
 select @SessionID        
        
 select SBranchID as ID,BranchName as Name from SBranchMaster        
  -- For Promote Next Year Session Name    
 Select SessionName as Name,SessionID as ID,SessionStatus as Extra1 from SessionMaster where SessionStartDate>@SelectedSessionEndDate and SBranchID=@SBranchID  order by sessionStartdate       
end
