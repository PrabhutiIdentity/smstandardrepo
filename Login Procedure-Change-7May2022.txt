alter proc [dbo].[sp_GetSBranches]  
(  
@UserID int=-1,  
@SBranchID int
)  
AS  
BEGIN  
if(@UserID<>-1)  
begin  
declare @SBranchID1 int  
select @SBranchID1 =SBranchID from LoginDetails where UserID=@UserID and SBranchID=@SBranchID and UserType in (0,8)  
Select SBranchID,BranchName,Logo from [dbo].[SBranchMaster] where SBranchID=@SBranchID1 
order by SBranchID  
end  
else  
begin   
Select SBranchID,BranchName,Logo from [dbo].[SBranchMaster]   
order by SBranchID  
end  
END  
 go 

alter PROCEDURE [dbo].[sp_GetUserByLoginName]              
(            
@LoginName nvarchar(50)            
)            
AS BEGIN              
 declare @UType int            
 declare @UID int   
 declare @SBranchID int  
select @UType=UserType,@UID=UserID,@SBranchID=SBranchID from LoginDetails where EmailID=@LoginName or UserName=@LoginName
--(CONVERT(nvarchar(50),UserID)= @LoginName or EmailID=@LoginName or UserName=@LoginName)             
            
 if(@UType=2)            
 --Accountant Login            
 begin            
  SELECT UserID,UserName, [Password],isnull(UserType,2) as RoleID,1 as IsApproved,'Account' as FullName,UD.SBranchID,            
  BM.Logo as BranchLogo,BM.StateID,BM.BranchName,BM.BranchSchoolName,BM.NotificationServerKey            
  FROM LoginDetails UD left outer join SBranchMaster BM on BM.SBranchID=UD.SBranchID where UserType=@UType and UserID=@UID  and UD.UserName=@LoginName          
 end            
 else if(@UType=3)            
 --Employee Login            
 begin            
  SELECT UserID,UserName, [Password],isnull(UserType,3) as RoleID,1 as IsApproved,EM.EmployeeName as FullName,EM.Photo as UserImage,UD.SBranchID,            
  BM.Logo as BranchLogo,BM.StateID,BM.BranchName,BM.BranchSchoolName,BM.NotificationServerKey            
  FROM LoginDetails UD left outer join EMployeeMaster EM on EM.EmployeeID=UD.UserID            
  left outer join SBranchMaster BM on BM.SBranchID=UD.SBranchID            
    where UD.UserType=@UType and UD.UserID=@UID  and EM.Status=1  and UD.UserName=@LoginName        
 end            
 else if(@UType=4)            
 --Parent Login            
 begin            
 declare @studentid int, @isblock int,@status int      
      
   select @studentid=ss.studentid, @isblock=isnull(ss.isblock,0),@status=ss.status from Student_Session ss    
   where studentid=(select studentid from studentmaster sm where sm.studentid=ss.studentid and ParentID=@UID and isnull(isblock,0)!=1)     
    
 --declare @studentid int, @isblock int,@status int        
          
 -- select @studentid=studentid, @isblock=isnull(isblock,0),@status=status from studentmaster sm     
 -- where parentid=@UID and studentid =(select studentid from Student_Session ss where ss.studentid=sm.studentid and isnull(isblock,0)!=1)     
    
  SELECT UserID,UserName, [Password],isnull(UserType,4) as RoleID,1 as IsApproved,EM.FatherName as FullName,EM.FatherImage as UserImage,UD.SBranchID,          
  BM.Logo as BranchLogo,BM.StateID,BM.BranchName,BM.BranchSchoolName,BM.NotificationServerKey          
  FROM LoginDetails UD left outer join ParentMaster EM on EM.ParentID=UD.UserID          
  left outer join SBranchMaster BM on BM.SBranchID=UD.SBranchID where UD.UserType=@UType and UD.UserID=@UID and ( @isblock!=1 or @status!=1)   and UD.UserName=@LoginName  
    
 -- select @studentid=studentid from studentmaster where parentid=@UID        
 --select @isblock=isnull(isblock,0),@status=status from student_session where studentid=@studentid        
 -- SELECT UserID,UserName, [Password],isnull(UserType,4) as RoleID,1 as IsApproved,EM.FatherName as FullName,EM.FatherImage as UserImage,UD.SBranchID,            
 -- BM.Logo as BranchLogo,BM.StateID,BM.BranchName,BM.BranchSchoolName,BM.NotificationServerKey            
 -- FROM LoginDetails UD left outer join ParentMaster EM on EM.ParentID=UD.UserID            
 -- left outer join SBranchMaster BM on BM.SBranchID=UD.SBranchID            
 --   where UD.UserType=@UType and UD.UserID=@UID and ( @isblock!=1 or @status!=1)           
 end            
 else if(@UType=5)            
 --Library Login            
 begin            
  SELECT UserID,UserName, [Password],isnull(UserType,5) as RoleID,1 as IsApproved,isnull(EM.EmployeeName,'Library') as FullName,EM.Photo as UserImage,UD.SBranchID,            
  BM.Logo as BranchLogo,BM.StateID,BM.BranchName,BM.BranchSchoolName,BM.NotificationServerKey            
  FROM LoginDetails UD left outer join EMployeeMaster EM on EM.EmployeeID=UD.UserID            
  left outer join SBranchMaster BM on BM.SBranchID=UD.SBranchID            
    where UD.UserType=@UType and UD.UserID=@UID  and UD.UserName=@LoginName          
 end            
 else if(@UType=6)      
 --Student Login            
 begin            
  SELECT UserID,UserName, [Password],isnull(UserType,6) as RoleID,1 as IsApproved,EM.Name as FullName,EM.Photo as UserImage,UD.SBranchID,            
  BM.Logo as BranchLogo,BM.StateID,BM.BranchName,BM.BranchSchoolName,BM.NotificationServerKey            
  FROM LoginDetails UD left outer join StudentMaster EM on EM.StudentID=UD.UserID            
  left outer join SBranchMaster BM on BM.SBranchID=UD.SBranchID            
    where UD.UserType=@UType and UD.UserID=@UID    and UD.UserName=@LoginName        
 end            
 else if(@UType=8)            
 --Principal Login            
 begin            
  SELECT UserID,UserName, [Password],isnull(UserType,6) as RoleID,1 as IsApproved,'Principal' as FullName,UD.SBranchID,            
 BM.Logo as BranchLogo,BM.StateID,BM.BranchName,BM.BranchSchoolName,BM.NotificationServerKey            
  FROM LoginDetails UD left outer join EMployeeMaster EM on EM.EmployeeID=UD.UserID            
  left outer join SBranchMaster BM on BM.SBranchID=UD.SBranchID            
    where UD.UserType=@UType and UD.UserID=@UID   and UD.UserName=@LoginName         
 end            
 else if(@UType=10)            
 --Reception Login            
 begin            
  SELECT UserID,UserName, [Password],isnull(UserType,10) as RoleID,1 as IsApproved,isnull(EM.EmployeeName,'Reception') as FullName,EM.Photo as UserImage,UD.SBranchID,          
  BM.Logo as BranchLogo,BM.StateID,BM.BranchName,BM.BranchSchoolName,BM.NotificationServerKey            
  FROM LoginDetails UD left outer join EMployeeMaster EM on EM.EmployeeID=UD.UserID            
  left outer join SBranchMaster BM on BM.SBranchID=UD.SBranchID            
    where UD.UserType=@UType and UD.UserID=@UID       and UD.UserName=@LoginName     
 end            
 else            
 begin            
 --declare @SBranchID int            
 --select @SBranchID=Min(SBranchID) from SBranchMaster            
  SELECT UserID,UserName, [Password],isnull(UserType,2) as RoleID,1 as IsApproved,            
  (isnull(UD.SBranchID,@SBranchID)) as SBranchID,BM.Logo as BranchLogo,BM.StateID,BM.BranchName,BM.BranchSchoolName,BM.NotificationServerKey            
  FROM LoginDetails UD left outer join EMployeeMaster EM on EM.EmployeeID=UD.UserID            
  left outer join SBranchMaster BM on BM.SBranchID=UD.SBranchID            
   where UD.EmailID=@LoginName or UserName=@LoginName            
  end            
END 
go
