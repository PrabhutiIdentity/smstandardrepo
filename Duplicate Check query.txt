
--Check Duplicate Record
select examid,studentid,marksscored,Grade,status,GradePoints,count(*)
from [dbo].[ExamResultMaster]
group by examid,studentid,marksscored,Grade,Status,GradePoints
having count(*)>1

--Delete Duplicate Record
WITH CTE(examid,studentid,marksscored,Grade,status,GradePoints,DuplicateCount)
AS (SELECT examid,studentid,marksscored,Grade,status,GradePoints,
           ROW_NUMBER() OVER(PARTITION BY examid,studentid,marksscored,Grade,status,GradePoints
           ORDER BY examid) AS DuplicateCount
    FROM [dbo].[ExamResultMaster] )
DELETE FROM CTE
WHERE DuplicateCount > 1;
