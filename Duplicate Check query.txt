
select ParamID,MainParamID,ParamName,ParamDesc,ParamType,SBranchID,count(*)
from [dbo].[PerformanceParameterMaster]
group by ParamID,MainParamID,ParamName,ParamDesc,ParamType,SBranchID
having count(*)>1

WITH CTE(ParamID,MainParamID,ParamName,ParamDesc,ParamType,SBranchID,DuplicateCount)
AS (SELECT ParamID,MainParamID,ParamName,ParamDesc,ParamType,SBranchID,
           ROW_NUMBER() OVER(PARTITION BY ParamID,MainParamID,ParamName,ParamDesc,ParamType,SBranchID
           ORDER BY ParamID) AS DuplicateCount
    FROM [dbo].[PerformanceParameterMaster] )
DELETE FROM CTE
WHERE DuplicateCount > 1;
