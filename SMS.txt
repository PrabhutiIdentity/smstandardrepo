create proc [dbo].[sp_InsertSMSSending]      
(      
 @SMSTemplateID int,      
 @SMSTypeID int,      
 @SMSSendDate datetime,      
 @TemplateText nvarchar(max),      
 @RecieverCatIDs nvarchar(50),      
 @Title nvarchar(200),      
 @Recievers ut_SMSProcessingLog READONLY,      
 @SBranchID int=1,      
 @content_id nvarchar(50)    
)      
AS      
BEGIN      
 declare @SMSID int      
 insert into SMSSendingMaster(SMSTemplateID,SMSTypeID ,SMSSendDate,TemplateText,RecieverCatIDs,Title,SBranchID,content_id)      
 values(@SMSTemplateID,@SMSTypeID ,@SMSSendDate,@TemplateText,@RecieverCatIDs,@Title,@SBranchID,@content_id)      
 select @SMSID=CAST(SCOPE_IDENTITY() as int)      
      
      
 INSERT INTO [SMSProcessingLog](SMSID, RecieverType, RecieverID,MobileNumber,content_id)      
    SELECT @SMSID, RecieverType, RecieverID,MobileNumber,content_id     
     FROM @Recievers      
 select @SMSID      
END 
go

alter proc [dbo].[sp_GetStudentPaymentSMSDetails]   
(    
@PaymentID int    
)    
AS    
BEGIN    
 Declare @MobileNo nvarchar(20)    
 Declare @Name nvarchar(50)    
 declare @ParentID int    
 Declare @NotificationSMSTo int    
 Declare @SMSTemplate nvarchar(max)    
 Declare @Amount nvarchar(20)    
 Declare @StudentID int    
 Declare @PaymentDate nvarchar(15)    
 Declare @FCMToken nvarchar(200)  
 Declare @Content_id nvarchar(50)  
 Declare @Sbranchid int

 select @Sbranchid=SBranchID from  PaymentMaster where PaymentID=@PaymentID  
    
 Select @Content_id=Content_id from MasterSettings where Type='FeePaymentNotificationSMSTemplate' and Sbranchid=@Sbranchid   
  Select @SMSTemplate=details from MasterSettings where Type='FeePaymentNotificationSMSTemplate' and Sbranchid=@Sbranchid  
 select @StudentID=PayeeID,@Amount=CONVERT(varchar, CAST(PaymentAmount AS money), 1),@PaymentDate=CONVERT(VARCHAR(11),PaymentDate,6)  from PaymentMaster where PaymentID=@PaymentID  
 Select @Name=Name,@ParentID=ParentID,@NotificationSMSTo=isnull(NotificationSMSTo,0),@MobileNo=GuardianMobileNo from StudentMaster where StudentID =@StudentID    and Sbranchid=@Sbranchid  
 select @FCMToken=FCMToken from AppUsers where UserType=4 and UserID=@ParentID    
 if(@SMSTemplate is null)    
 begin    
  set @SMSTemplate='Dear [Reciever],%0a Fee Payment of [Amount] Rs for [StudentName] is Recieved on [PaymentDate], Thanks.'    
 end    
    
 if(@NotificationSMSTo=0)    
 begin    
  Select @MobileNo=[FatherMobileNo] from ParentMaster where ParentID=@ParentID    
 end    
 else if(@NotificationSMSTo=1)    
 begin    
  Select @MobileNo=[MotherMobileNo] from ParentMaster where ParentID=@ParentID    
 end    
 set @SMSTemplate=replace(@SMSTemplate, '[Reciever]', 'Parent')    
 set @SMSTemplate=replace(@SMSTemplate, '[Amount]', @Amount)    
 set @SMSTemplate=replace(@SMSTemplate, '[StudentName]', @Name)    
 set @SMSTemplate=replace(@SMSTemplate, '[PaymentDate]', @PaymentDate)    
    
 select @SMSTemplate as SMSText,@MobileNo as Number,@FCMToken as FCMToken,@ParentID as RecieverID, 4 as RecieverType,@Content_id as Content_id    
END    
  go
