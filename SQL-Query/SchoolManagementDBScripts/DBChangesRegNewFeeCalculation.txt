Create Function [dbo].[fn_GetStudentSimpleTransportAmount] 
(
@StudentID int=457,
@LYear int=2019,
@UserType int=0,
@LMonth int=5,
@HorT int=0
)
Returns numeric(10,2)
as
begin
	Declare @MStartDate date=cast(Cast(@LYear as nvarchar(4))+'-'+ Cast(@LMonth as nvarchar(4))+'-1' as date)
	Declare @MEndDate date=EOMonth(@MStartDate)

	Declare @Amount numeric(18,2)=0
	if(@HorT=0)
	begin
	Select @Amount= isnull(sum(Amount*(Days+1)/TotalDays),0) from (
		select StartDate,EndDate,KeyID,ChangeType,DATEDIFF(DAY, StartDate,isnull(EndDate,@MEndDate)) as Days,datepart(day,@MEndDate) as TotalDays,
		(select isnull(Rate,0) from TransportRouteDetails TRM where TRM.StopID=KeyID) as Amount from 
		(select (case when  StartDate is null or StartDate<@MStartDate then @MStartDate else StartDate end) as StartDate,
		(case when EndDate is null or EndDate>@MEndDate then @MEndDate else EndDate end) as EndDate
		, KeyID,ChangeType
		from TransportHostalAllocationDelocation where UserType=@UserType and ChangeType=0 and UserID=@StudentID and (StartDate<EndDate or EndDate is null)
		and ((StartDate between @MStartDate and @MEndDate or StartDate<@MStartDate) and (isnull(EndDate,@MEndDate) between @MStartDate and @MEndDate or EndDate>@MEndDate))
		 )t)tt
	end
	else
	begin
		Select @Amount= isnull(sum(Amount*(Days+1)/TotalDays),0) from (
					select StartDate,EndDate,KeyID,ChangeType,DATEDIFF(DAY, StartDate,EndDate) as Days,datepart(day,@MEndDate) as TotalDays,
					(select isnull(Rate,0) from Hostel_RoomTypeRateMaster RTRM where RTRM.ID=(select RoomTypeID from Hostel_RoomMaster RM where RM.ID=KeyID)) as Amount
						 from 	(select (case when StartDate is null or StartDate<@MStartDate then @MStartDate else StartDate end) as StartDate,
					(case when EndDate is null or EndDate>@MEndDate then @MEndDate else EndDate end) as EndDate
					, KeyID,ChangeType
					from TransportHostalAllocationDelocation where UserType=@UserType and ChangeType=1 and UserID=@StudentID and (StartDate<EndDate or EndDate is null)
					and ((StartDate between @MStartDate and @MEndDate or StartDate<@MStartDate) and (isnull(EndDate,@MEndDate) between @MStartDate and @MEndDate or EndDate>@MEndDate))
					 )t)tt
	end
	-- select * from TransportHostalAllocationDelocation where UserType=0 and UserID=@StudentID and ChangeType=0 and
	-- (StartDate between @MStartDate and @MEndDate or isnull(EndDate,(case when StartDate<=@MStartDate then @MEndDate else null end)) between @MStartDate and @MEndDate or
	-- @MStartDate Between StartDate and isnull(EndDate,(case when StartDate<=@MStartDate then @MEndDate else null end)) or 
	-- @MEndDate Between StartDate and isnull(EndDate,(case when StartDate<=@MStartDate then @MEndDate else null end)))
	Return @Amount
END

GO

Create Procedure dbo.spn_GetStudentFeeDetailsSingle
as
BEGIN	
	Declare @StudentID int=41
	Declare @ClassID int=1
	Declare @GroupID int=1
	Declare @SBranchID int=1
	Declare @SessionID int=1
	Declare @QuotaID int=1
	Declare @Month int=5
	Declare @Year int=2019
	Declare @QDate date='2019-05-22'
	Declare @FeePaymentMode int
	Declare @IsAdmissionFee int=1
	Declare @SectionID int
	Declare @StartDate date='2019-04-01'
	Declare @EndDate date='2020-03-31'

	Declare @BaseDate date

	Declare @MonthType int

	Declare @LMonth int
	Declare @LYear int

	Declare @TransportFeeMode int
	Declare @HostelFeeMode int
	Declare @LastPayDay int
	Declare @StudentSessionDate date
	Declare @StudentSessionEndDate date
	Declare @SessionStartDate date
	Declare @StopID int
	declare @IsTransport int
	declare @IsHostel int
	
	Select @IsTransport=StopID,@IsHostel=HostelRoomID,@StopID=StopID from StudentMaster where StudentID=@StudentID

	if(@IsTransport<>0)
	begin
		Select @TransportFeeMode=details from MasterSettings where Type='TransportFeeMode' and SBranchID=@SBranchID
	end
	if(@IsHostel<>0)
	begin
		Select @HostelFeeMode=details from MasterSettings where Type='HostelFeeMode' and SBranchID=@SBranchID
	end
	Select @LastPayDay=details from MasterSettings where Type='FeePaymentReminderDate' and SBranchID=@SBranchID

	select @SessionStartDate=SessionStartDate from SessionMaster where SessionID=@SessionID

	Select @StudentSessionDate=FromDate,@StudentSessionEndDate=ToDate,@ClassID=ClassID,@QuotaID=QuotaID,
	@SectionID=SectionID,@SessionID=SessionID,@FeePaymentMode=FeePaymentMode,@SBranchID=SBranchID ,@IsAdmissionFee=isnull(IsAdmissionFeeApplicable,0)
	from Student_Session where StudentID=@StudentID and SessionID=@SessionID

	if(@SessionStartDate>@StudentSessionDate)
	begin
		set @StudentSessionDate=@SessionStartDate
	end
	if(@EndDate>@StudentSessionEndDate)
	begin
		set @EndDate=@StudentSessionEndDate
	end
	
	set @StartDate=@StudentSessionDate
	set @BaseDate=@StartDate

	Select @GroupID=GroupID from Class_Sections where ID=@SectionID


	Declare @FeeTypeApplicables table(FeeTypeID int)


	Declare @FinalPaymentDetails table(Month int,Year int,FeeTypeID int,FeeTypeName nvarchar(50),FeeTypeApplicable int,FeeMonths nvarchar(50),FeeAmount numeric(18,2),DiscPer numeric(10,2),
	Status int,PAmount numeric(18,2),PDiscAmt numeric(18,2),NetApplicablePayment numeric(18,2),PaymentRecieved numeric(18,2),PaymentDate date,ApprovedAmount numeric(18,2))

	--Discount Request Handeling
	Declare @Discounts table(StudentID int,FeeTypeID int,FeeMonth int,FeeYear int,ApprovedAmount decimal(18,2))
	Insert into @Discounts Select StudentID,FeeTypeID,FeeMonth,FeeYear,ApprovedAmount 
	from [FeeDiscountRequestDetails] FDD left outer join [FeeDiscountRequestMaster] FDM on FDM.DiscRequestID=FDD.DiscRequestID 
	where StudentID=@StudentID and Status=1

	Declare @FeeStructureTable table(FeeTypeID int,FeeTypeName nvarchar(50),FeeTypeApplicable int,FeeMonths nvarchar(50),
	FeeAmount numeric(10,2),DiscPer numeric(10,2),Status int)

	Declare @FeeDetailsTable table(FeeTypeID int,FeeTypeName nvarchar(50),FeeTypeApplicable int,FeeMonths nvarchar(50),
	FeeAmount numeric(10,2),DiscPer numeric(10,2),Status int,Month int,Year int,PAmount numeric(18,2),PDiscAmt numeric(18,2),
	NetApplicablePayment numeric(18,2),PaymentRecieved numeric(18,2),PaymentDate date,RDiscAmt numeric(18,2))

	insert into @FeeStructureTable
	Select FTM.FeeTypeID,FeeTypeName,FTM.FeeTypeApplicable,FTM.FeeMonths,CFS.FeeAmount,QDD.DiscPer,CFS.Status
	from FeeTypeMaster FTM left Outer Join ClassFeeStructureMaster CFS on CFS.FeeTypeID=FTM.FeeTypeID
	left outer join QuotaDiscountDetails QDD on QDD.FeeTypeID=FTM.FeeTypeID
	where  CFS.ClassID=@ClassID and CFS.GroupID=@GroupID and FTM.SBranchID=@SBranchID and CFS.SessionID=@SessionID and QDD.QuotaID=@QuotaID and QDD.SessionID=@SessionID

	while(@StartDate<@EndDate)
	begin
		delete from @FeeTypeApplicables
		set @LMonth=Datepart(month,@StartDate)
		set @LYear=Datepart(Year,@StartDate)

		set @MonthType=(Case when datepart(month,@BaseDate)=@LMonth and (datepart(month,dateadd(month,3,@SessionStartDate))=@LMonth  or datepart(month,dateadd(month,9,@SessionStartDate))=@LMonth) then 5 else
			(Case when datepart(month,@BaseDate)=@LMonth and (datepart(month,dateadd(month,6,@SessionStartDate))=@LMonth) then 6 else 
			(case when datepart(month,@BaseDate)=@LMonth then (case when @IsAdmissionFee=0 then 1 else 0 end)  
			else (Case when datepart(month,dateadd(month,3,@SessionStartDate))=@LMonth  or datepart(month,dateadd(month,9,@SessionStartDate))=@LMonth then 3
			else (case when  datepart(month,dateadd(month,6,@SessionStartDate))=@LMonth then 4 else 2 end) end) end)end)end)

		if(@MonthType=0)
		begin
			Insert into @FeeTypeApplicables values(1)
			Insert into @FeeTypeApplicables values(2)
			Insert into @FeeTypeApplicables values(3)
			Insert into @FeeTypeApplicables values(4)
			Insert into @FeeTypeApplicables values(5)
			Insert into @FeeTypeApplicables values(8)
		end
		else if(@MonthType=1)
		begin
			Insert into @FeeTypeApplicables values(1)
			Insert into @FeeTypeApplicables values(2)
			Insert into @FeeTypeApplicables values(3)
			Insert into @FeeTypeApplicables values(4)
			Insert into @FeeTypeApplicables values(5)
		end
		else if(@MonthType=2)
		begin
			Insert into @FeeTypeApplicables values(2)
			Insert into @FeeTypeApplicables values(5)
		end
		else if(@MonthType=3)
		begin
			Insert into @FeeTypeApplicables values(2)
			Insert into @FeeTypeApplicables values(3)
			Insert into @FeeTypeApplicables values(5)
		end
		else if(@MonthType=4)
		begin
			Insert into @FeeTypeApplicables values(2)
			Insert into @FeeTypeApplicables values(3)
			Insert into @FeeTypeApplicables values(4)
			Insert into @FeeTypeApplicables values(5)
		end
		else if(@MonthType=5)
		begin
			Insert into @FeeTypeApplicables values(1)
			Insert into @FeeTypeApplicables values(2)
			Insert into @FeeTypeApplicables values(3)
			Insert into @FeeTypeApplicables values(4)
			Insert into @FeeTypeApplicables values(5)
		end	
		else if(@MonthType=6)
		begin
			Insert into @FeeTypeApplicables values(1)
			Insert into @FeeTypeApplicables values(2)
			Insert into @FeeTypeApplicables values(3)
			Insert into @FeeTypeApplicables values(4)
			Insert into @FeeTypeApplicables values(5)
		end
		insert into @FinalPaymentDetails
		select @LMonth as Month,@LYear as Year,FST.FeeTypeID,FeeTypeName,FeeTypeApplicable,FeeMonths,
		FeeAmount*(case when FeeTypeApplicable=2 then @FeePaymentMode else 1 end) as FeeAmount,
		DiscPer,Status,PAmount,PDIscAmt,NetApplicablePayment,PaymentRecieved,PaymentDate, ApprovedAmount
		from  @FeeStructureTable FST left outer join 
		(select FeeTypeID,Sum(Amount) as PAmount,sum(DiscAmt) as PDiscAmt,Sum(NetApplicablePayment) as NetApplicablePayment,Sum(PaymentRecieved) as PaymentRecieved,Min(PaymentDate) as PaymentDate
		 from PaymentDetails where Month=@LMonth and Year=@LYear and PayeeID=@StudentID 
		 group by FeeTypeID)p on P.FeeTypeID=FST.FeeTypeID left outer join @Discounts D on D.FeeTypeID=FST.FeeTypeID and  FeeMonth=@LMonth and FeeYear=@LYear
		 where FeeTypeApplicable in (Select FeeTypeID from @FeeTypeApplicables) and FeeTypeApplicable not in (Case when isnull(@IsTransport,0)=0 then 5 else 0 end) 
		 and FeeTypeApplicable not in (Case when isnull(@IsHostel,0)=0 then 6 else 0 end) 	
		 and FST.FeeTypeID not in (case when (PaymentDate is null and datepart(year,@QDate)*12+datepart(month,@QDate)>@LYear*12+@LMonth) 
		 or PaymentDate is not null and ((datepart(year,isnull(PaymentDate,@QDate))*12+datepart(month,isnull(PaymentDate,@QDate))>@LYear*12+@LMonth or 
		 datepart(year,isnull(PaymentDate,@QDate))*12+datepart(month,isnull(PaymentDate,@QDate))=@LYear*12+@LMonth and datepart(day,@QDate)>@LastPayDay)) then 0 else -2 end) 	

		 set @StartDate=dateadd(month,1,@StartDate)
	end
	select * from @FinalPaymentDetails
END
